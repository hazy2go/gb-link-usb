cmake_minimum_required(VERSION 3.13)
# GBLINK_BOARD: breadboard (Pico 2 W, GP0-2) or revA (our PCB: RP2350A minimal core, GP11-14, LED GP21)
set(GBLINK_BOARD breadboard CACHE STRING "breadboard or revA")
if(GBLINK_BOARD STREQUAL "revA")
  # RP2350A + 12 MHz ABM8-272-T3 crystal + QSPI flash like the Pico 2; rev A has a 16 MB W25Q128
  set(PICO_BOARD pico2 CACHE STRING "Board")
  add_compile_definitions(PICO_FLASH_SIZE_BYTES=16777216)
else()
  set(PICO_BOARD pico2_w CACHE STRING "Board")
endif()
include(pico_sdk_import.cmake)
project(gb_link_usb C CXX ASM)
pico_sdk_init()

add_executable(gb_link_usb src/main.c src/printer.c)
pico_generate_pio_header(gb_link_usb ${CMAKE_CURRENT_LIST_DIR}/src/gb_link.pio)
target_include_directories(gb_link_usb PRIVATE src)
target_compile_options(gb_link_usb PRIVATE -Wall -Wextra)
target_link_libraries(gb_link_usb pico_stdlib pico_multicore hardware_pio)
if(GBLINK_BOARD STREQUAL "revA")
  target_compile_definitions(gb_link_usb PRIVATE GBLINK_BOARD_REVA=1)
elseif(GBLINK_BOARD STREQUAL "breadboard")
  target_link_libraries(gb_link_usb pico_cyw43_arch_none)
else()
  message(FATAL_ERROR "GBLINK_BOARD must be breadboard or revA")
endif()
pico_enable_stdio_usb(gb_link_usb 1)
pico_enable_stdio_uart(gb_link_usb 0)
pico_set_program_name(gb_link_usb "gb-link-usb printer")
pico_add_extra_outputs(gb_link_usb)

# Debug: -DDEBUG_FIXED_REPLY=0xA5 answers that byte to everything (tests the TX path).
if(DEBUG_FIXED_REPLY)
  target_compile_definitions(gb_link_usb PRIVATE DEBUG_FIXED_REPLY=${DEBUG_FIXED_REPLY})
endif()
