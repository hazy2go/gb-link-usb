cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2_w CACHE STRING "Board")
include(pico_sdk_import.cmake)
project(gb_link_usb C CXX ASM)
pico_sdk_init()

add_executable(gb_link_usb src/main.c src/printer.c)
pico_generate_pio_header(gb_link_usb ${CMAKE_CURRENT_LIST_DIR}/src/gb_link.pio)
target_include_directories(gb_link_usb PRIVATE src)
target_compile_options(gb_link_usb PRIVATE -Wall -Wextra)
target_link_libraries(gb_link_usb pico_stdlib pico_multicore hardware_pio pico_cyw43_arch_none)
pico_enable_stdio_usb(gb_link_usb 1)
pico_enable_stdio_uart(gb_link_usb 0)
pico_set_program_name(gb_link_usb "gb-link-usb printer")
pico_add_extra_outputs(gb_link_usb)

# Debug: -DDEBUG_FIXED_REPLY=0xA5 answers that byte to everything (tests the TX path).
if(DEBUG_FIXED_REPLY)
  target_compile_definitions(gb_link_usb PRIVATE DEBUG_FIXED_REPLY=${DEBUG_FIXED_REPLY})
endif()
